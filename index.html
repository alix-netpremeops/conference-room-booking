// Room booking — Slack reminders, confirmations and live schedule canvas. // // bookingTick every minute: DM bookers 10 min before, auto-release // unconfirmed bookings, keep the canvas "Right now" current // slackInteractions Slack interactivity endpoint for Confirm / Release buttons // onBookingChange refreshes the canvas whenever a booking is written const { onSchedule } = require('firebase-functions/v2/scheduler'); const { onRequest } = require('firebase-functions/v2/https'); const { onValueWritten } = require('firebase-functions/v2/database'); const { defineSecret, defineString } = require('firebase-functions/params'); const logger = require('firebase-functions/logger'); const admin = require('firebase-admin'); const crypto = require('crypto'); admin.initializeApp(); const db = admin.database(); const SLACK_BOT_TOKEN = defineSecret('SLACK_BOT_TOKEN'); const SLACK_SIGNING_SECRET = defineSecret('SLACK_SIGNING_SECRET'); const SLACK_CHANNEL_ID = defineString('SLACK_CHANNEL_ID'); const TIME_ZONE = defineString('TIME_ZONE', { default: 'America/Los_Angeles' }); const BOOKING_URL = defineString('BOOKING_URL', { default: '' }); const ROOMS = ['205', '202']; const REMINDER_MIN = 10; // DM the booker this many minutes before start const RELEASE_AFTER_MIN = 5; // release if still unconfirmed this long after start const MIN = 60 * 1000; // ── Slack helpers ──────────────────────────────────────────────────────────── async function slack(method, body) { const form = new URLSearchParams(); for (const [k, v] of Object.entries(body)) { if (v !== undefined) form.set(k, typeof v === 'string' ? v : JSON.stringify(v)); } const res = await fetch(`https://slack.com/api/${method}`, { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded', Authorization: `Bearer ${SLACK_BOT_TOKEN.value()}`, }, body: form, }); const data = await res.json(); if (!data.ok) throw new Error(`${method}: ${data.error}`); return data; } function verifySlack(req) { const ts = req.get('x-slack-request-timestamp'); const sig = req.get('x-slack-signature'); if (!ts || !sig || Math.abs(Date.now() / 1000 - Number(ts)) > 300) return false; const mine = 'v0=' + crypto .createHmac('sha256', SLACK_SIGNING_SECRET.value()) .update(`v0:${ts}:${req.rawBody.toString('utf8')}`) .digest('hex'); return mine.length === sig.length && crypto.timingSafeEqual(Buffer.from(mine), Buffer.from(sig)); } const mrkdwn = s => String(s ?? '').replace(/&/g, '&').replace(//g, '>'); const fmtTime = ms => new Intl.DateTimeFormat('en-US', { timeZone: TIME_ZONE.value(), hour: 'numeric', minute: '2-digit' }).format(ms); const fmtDay = ms => new Intl.DateTimeFormat('en-US', { timeZone: TIME_ZONE.value(), weekday: 'long', month: 'short', day: 'numeric' }).format(ms); const dayKey = ms => new Intl.DateTimeFormat('en-CA', { timeZone: TIME_ZONE.value(), year: 'numeric', month: '2-digit', day: '2-digit' }).format(ms); const span = b => `${fmtTime(b.startAt)} – ${fmtTime(b.endAt)}`; async function slackUserFor(b) { if (!b.organizerEmail) return null; try { return (await slack('users.lookupByEmail', { email: b.organizerEmail })).user.id; } catch (e) { logger.warn(`No Slack user for ${b.organizerEmail}`, e.message); return null; } } // Update fields only if the booking still exists (a plain update() would // resurrect a deleted booking as a partial node). async function updateIfExists(id, fields) { const r = await db.ref(`bookings/${id}`).transaction(cur => (cur === null ? null : { ...cur, ...fields })); return r.snapshot.val(); } async function updateReminderMessage(b, text) { if (!b.reminderChannel || !b.reminderTs) return; try { await slack('chat.update', { channel: b.reminderChannel, ts: b.reminderTs, text, blocks: [] }); } catch (e) { logger.warn('chat.update failed', e.message); } } // ── Reminders & auto-release ───────────────────────────────────────────────── function reminderBlocks(id, b, mention) { return [ { type: 'section', text: { type: 'mrkdwn', text: `:alarm_clock: ${mention}*Room ${b.room}* starts in ${REMINDER_MIN} minutes\n` + `*${mrkdwn(b.title)}* · ${span(b)}\n` + `Please confirm you still need it. If not confirmed by ${fmtTime(b.startAt + RELEASE_AFTER_MIN * MIN)}, ` + `the room will be released for others.`, }, }, { type: 'actions', elements: [ { type: 'button', style: 'primary', text: { type: 'plain_text', text: 'Confirm' }, action_id: 'confirm_booking', value: id }, { type: 'button', style: 'danger', text: { type: 'plain_text', text: 'Release room' }, action_id: 'release_booking', value: id }, ], }, ]; } async function sendReminder(id, b) { const userId = await slackUserFor(b); const text = `Room ${b.room} starts in ${REMINDER_MIN} minutes — please confirm your booking`; // Prefer a DM (always notifies). Fall back to the channel, naming the booker. const channel = userId || SLACK_CHANNEL_ID.value(); const mention = userId ? '' : `*${mrkdwn(b.organizer || 'Someone')}*, `; const msg = await slack('chat.postMessage', { channel, text, blocks: reminderBlocks(id, b, mention) }); await updateIfExists(id, { slackUserId: userId || null, reminderChannel: msg.channel, reminderTs: msg.ts }); } async function handleBooking(id, b, now) { if (b.status !== 'pending' || !b.startAt) return; if (!b.reminderSentAt && b.startAt - now <= REMINDER_MIN * MIN && now < b.startAt + RELEASE_AFTER_MIN * MIN) { // Claim the reminder atomically so overlapping ticks never double-send. const r = await db.ref(`bookings/${id}`).transaction(cur => { if (cur === null) return null; if (cur.status !== 'pending' || cur.reminderSentAt) return; return { ...cur, reminderSentAt: now }; }); if (!r.committed || r.snapshot.val()?.reminderSentAt !== now) return; try { await sendReminder(id, b); } catch (e) { // Un-claim so the next tick retries; never auto-release a booking whose owner wasn't reminded. logger.error(`Reminder failed for ${id}`, e.message); await updateIfExists(id, { reminderSentAt: null }); } return; } if (b.reminderSentAt && now >= b.startAt + RELEASE_AFTER_MIN * MIN) { const r = await db.ref(`bookings/${id}`).transaction(cur => { if (cur === null) return null; if (cur.status !== 'pending') return; // confirmed in the meantime return null; }); if (!r.committed || r.snapshot.val() !== null) return; await updateReminderMessage(b, `:unlock: Not confirmed in time — Room ${b.room} (${span(b)}) was released.`); await slack('chat.postMessage', { channel: SLACK_CHANNEL_ID.value(), text: `:unlock: Room ${b.room} is free ${span(b)} — booking “${b.title}” by ${b.organizer || 'someone'} wasn't confirmed.`, }).catch(e => logger.warn('Release notice failed', e.message)); } } exports.bookingTick = onSchedule( { schedule: 'every 1 minutes', secrets: [SLACK_BOT_TOKEN] }, async () => { const now = Date.now(); const snap = await db.ref('bookings').orderByChild('startAt') .startAt(now - 2 * 60 * MIN).endAt(now + (REMINDER_MIN + 1) * MIN).get(); const jobs = []; snap.forEach(child => { jobs.push(handleBooking(child.key, child.val(), now)); }); const results = await Promise.allSettled(jobs); results.filter(r => r.status === 'rejected').forEach(r => logger.error(r.reason)); try { await refreshCanvas(); } catch (e) { logger.error('Canvas refresh failed', e.message); } } ); // ── Slack button clicks ────────────────────────────────────────────────────── exports.slackInteractions = onRequest( { secrets: [SLACK_BOT_TOKEN, SLACK_SIGNING_SECRET] }, async (req, res) => { if (req.method !== 'POST' || !verifySlack(req)) { res.status(401).send('Invalid signature'); return; } const payload = JSON.parse(req.body.payload || '{}'); const action = payload.actions?.[0]; if (!action) { res.sendStatus(200); return; } const id = action.value; const clicker = payload.user?.id; const reply = body => fetch(payload.response_url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body), }); const b = (await db.ref(`bookings/${id}`).get()).val(); if (!b) { await reply({ replace_original: true, text: 'This booking no longer exists — it was cancelled or released.' }); res.sendStatus(200); return; } if (b.slackUserId && b.slackUserId !== clicker) { await reply({ response_type: 'ephemeral', replace_original: false, text: `Only ${b.organizer} can respond to this booking.` }); res.sendStatus(200); return; } if (action.action_id === 'confirm_booking') { const after = await updateIfExists(id, { status: 'confirmed', confirmedAt: Date.now() }); await reply({ replace_original: true, text: after ? `:white_check_mark: Confirmed — Room ${b.room} is yours, ${span(b)} (“${b.title}”).` : 'This booking no longer exists — it was cancelled or released.', }); } else if (action.action_id === 'release_booking') { await db.ref(`bookings/${id}`).remove(); await reply({ replace_original: true, text: `:unlock: Released Room ${b.room} (${span(b)}). Thanks for freeing it up!` }); await slack('chat.postMessage', { channel: SLACK_CHANNEL_ID.value(), text: `:unlock: Room ${b.room} just opened up ${span(b)} — ${b.organizer || 'the booker'} released it.`, }).catch(e => logger.warn('Release notice failed', e.message)); } res.sendStatus(200); } ); // ── Live schedule canvas (channel tab) ─────────────────────────────────────── const clean = s => String(s ?? '').replace(/[*_~`#<>\[\]|!]/g, '').trim() || 'Untitled'; function statusTag(b) { if (b.status === 'confirmed') return ' ✅'; if (b.status === 'pending' && b.reminderSentAt) return ' ⏳ awaiting confirmation'; return ''; } function buildCanvas(list, now) { const today = dayKey(now); const lines = ['# Room schedule', '']; if (BOOKING_URL.value()) lines.push(`[Book a room](${BOOKING_URL.value()})`, ''); lines.push('## Right now'); for (const room of ROOMS) { const upcoming = list.filter(b => b.room === room && b.endAt > now); const cur = upcoming.find(b => b.startAt <= now); const next = upcoming.find(b => b.startAt > now && dayKey(b.startAt) === today); if (cur) { lines.push(`- **Room ${room}** 🔴 In use — ${clean(cur.title)} · booked by ${clean(cur.organizer)} · until ${fmtTime(cur.endAt)}`); } else { lines.push(`- **Room ${room}** 🟢 Free ${next ? `until ${fmtTime(next.startAt)} (${clean(next.organizer)})` : 'for the rest of the day'}`); } } const days = [...new Set(list.map(b => dayKey(b.startAt)))].filter(k => k > today).sort(); for (const k of [today, ...days]) { const dayList = list.filter(b => dayKey(b.startAt) === k); lines.push('', `## ${k === today ? `Today — ${fmtDay(now)}` : fmtDay(dayList[0].startAt)}`); for (const room of ROOMS) { const rb = dayList.filter(b => b.room === room); lines.push(`### Room ${room}`); if (!rb.length) lines.push('_No bookings_'); for (const b of rb) { lines.push(`- ${span(b)} · **${clean(b.title)}** · ${clean(b.organizer)}${statusTag(b)}`); } } } lines.push('', `_Times in ${TIME_ZONE.value()}. This tab updates automatically — edits here will be overwritten._`); return lines.join('\n'); } async function refreshCanvas() { const now = Date.now(); const snap = await db.ref('bookings').orderByChild('startAt') .startAt(now - 24 * 60 * MIN).endAt(now + 7 * 24 * 60 * MIN).get(); const list = []; snap.forEach(c => { const b = c.val(); if (ROOMS.includes(b.room) && b.endAt) list.push(b); }); list.sort((a, b) => a.startAt - b.startAt); const today = dayKey(now); const markdown = buildCanvas(list.filter(b => dayKey(b.startAt) >= today), now); const hash = crypto.createHash('sha1').update(markdown).digest('hex'); const state = (await db.ref('slack').get()).val() || {}; if (state.canvasId && state.canvasHash === hash) return; // nothing changed const document_content = { type: 'markdown', markdown }; if (state.canvasId) { await slack('canvases.edit', { canvas_id: state.canvasId, changes: [{ operation: 'replace', document_content }] }); } else { // Creates a new canvas tabbed in the channel (doesn't touch an existing channel canvas). const channel = SLACK_CHANNEL_ID.value(); const r = await slack('canvases.create', { title: 'Room schedule', document_content, channel_id: channel }); await slack('canvases.access.set', { canvas_id: r.canvas_id, access_level: 'read', channel_ids: [channel] }) .catch(e => logger.warn('canvases.access.set failed', e.message)); await db.ref('slack/canvasId').set(r.canvas_id); } await db.ref('slack/canvasHash').set(hash); } exports.onBookingChange = onValueWritten( { ref: '/bookings/{id}', secrets: [SLACK_BOT_TOKEN] }, async () => { try { await refreshCanvas(); } catch (e) { logger.error('Canvas refresh failed', e.message); } } );
